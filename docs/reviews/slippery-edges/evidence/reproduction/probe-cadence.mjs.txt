const rows=[];for(const dt of [1/120,1/60,.025,.05,.1]){const c=vehicleYardContent({sourcePin:'0000000000000000000000000000000000000000'}),v=createVehicle({id:'compact',position:{x:30-(.985*Math.cos(.15)+1.85*Math.sin(.15)),y:10,h:0},headingRad:.15});c.surfaces=c.surfaces.filter(s=>s.supports);c.surfaces.push({id:'glance',floorId:'floor-0',role:'solid',bounds:{min:{x:30,y:0,h:0},max:{x:30.02,y:32,h:2}},supports:false,supportFloorId:null});c.vehicles=[{id:v.id,body:v.body,position:v.position,headingRad:v.headingRad}];c.actors=[{...c.actors[0],position:transform(v,-1.45,.35)}];c.resources={player:c.resources.player};const h=createVehicleYard(c,{collisionProfile:'oriented-slide-v1',movementProfile:'disc-slide-v1'});const s=h.connector;const send=(type,more)=>s.dispatch({version:1,type,actorId:'player',expected:s.readForActor('player').expected,...more});if(!send('board',{vehicleId:'compact',seatId:'driver'}).ok||!send('control',{owner:'probe',held:true,throttle:1,steering:0}).ok)throw Error('fixture');for(let n=0;n<Math.round(3/dt);n++)h.advance(dt);const end=s.read().vehicles[0];rows.push({dt,distanceM:Math.hypot(end.position.x-v.position.x,end.position.y-v.position.y),speedMps:end.speedMps});h.dispose();}console.log(JSON.stringify(rows,null,2));
