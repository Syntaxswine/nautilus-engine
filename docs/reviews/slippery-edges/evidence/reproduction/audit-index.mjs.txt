import fs from 'node:fs';import path from 'node:path';import assert from 'node:assert/strict';import {createHash} from 'node:crypto';import {execFileSync} from 'node:child_process';
const root=process.cwd(),sha=b=>createHash('sha256').update(b).digest('hex'),manifest='docs/reviews/slippery-edges/registration.json',guard='tools/traversal-registration.mjs',m=JSON.parse(fs.readFileSync(manifest)),files=[...m.files.map(r=>r.path),manifest,guard];
const batch=execFileSync('git',['cat-file','--batch'],{cwd:root,input:files.map(p=>':'+p).join('\n')+'\n',windowsHide:true,maxBuffer:32*1024*1024});let cursor=0,total=0;const indexed=new Map();
for(const file of files){const end=batch.indexOf(10,cursor),header=batch.subarray(cursor,end).toString(),size=Number(header.split(' ')[2]);assert.ok(Number.isFinite(size)&&header.includes(' blob '),header);cursor=end+1;const b=batch.subarray(cursor,cursor+size);cursor+=size+1;assert.ok(b.equals(fs.readFileSync(path.join(root,file))),'Index differs from worktree: '+file);indexed.set(file,b);total+=size;}
for(const r of m.files){const b=indexed.get(r.path);assert.equal(sha(b),r.sha256);assert.equal(b.length,r.bytes);}
const donor=JSON.parse(fs.readFileSync(m.donorManifest));for(const row of donor.files){const b=indexed.get(row.path);assert.equal(sha(b),row.sha256);assert.equal(createHash('sha1').update('blob '+b.length+'\0').update(b).digest('hex'),row.gitBlob);}
const notice=execFileSync('git',['show',m.base+':NOTICE.md'],{windowsHide:true});assert.ok(notice.equals(fs.readFileSync('NOTICE.md')));
const report={status:'passed',base:m.base,indexRecords:files.length,indexBytes:total,allCurrentIndexBytesMatchWorktreeAndLedger:true,originalGitBlobs:donor.files.length,originalBytes:donor.files.reduce((n,r)=>n+r.bytes,0),rootNoticeRetained:true,registrationSHA256:sha(indexed.get(manifest))};fs.writeFileSync('artifacts/traversal/index-audit.json',JSON.stringify(report,null,2)+'\n');console.log(JSON.stringify(report));
