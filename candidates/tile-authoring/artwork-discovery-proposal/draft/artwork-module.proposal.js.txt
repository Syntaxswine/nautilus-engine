function exact(pin){
 if(!pin||![Object.prototype,null].includes(Object.getPrototypeOf(pin)))throw TypeError('Use an exact Core artwork pin.');
 const d=Object.getOwnPropertyDescriptors(pin),keys=Reflect.ownKeys(d);
 if(keys.length!==3||keys.some(k=>!['id','revision','contentDigest'].includes(k)||!d[k].enumerable||!Object.hasOwn(d[k],'value')))throw TypeError('Use exactly id,revision,contentDigest; no alias/latest lookup.');
 const e=ENTRIES.find(e=>e.record.id===d.id.value&&e.record.revision===d.revision.value&&e.record.contentDigest===d.contentDigest.value);
 if(!e)throw RangeError('Unknown exact shared artwork pin.');return e;
}
