import {createTileAuthoringCommands,finiteUIData,tileAuthoringExpected,tileAuthoringPin,tileAuthoringLabelMessage,TILE_AUTHORING_FORM_LIMITS} from './commands.js';

const serviceMethods = ['describe','snapshot','query','prepare','apply','cancel','recover'];
const issue = (code, message) => Object.assign(new TypeError(code + ': ' + message), {code});
const enabled = (view, name) => view?.presentation.actions[name]?.enabled === true;
const ownFunction = (object, key) => !!object && typeof Object.getOwnPropertyDescriptor(object, key)?.value === 'function';
const stringify = value => JSON.stringify(value);
const sameData = (a,b) => a===b || !!a && !!b && typeof a==='object' && typeof b==='object' && Array.isArray(a)===Array.isArray(b) && Object.keys(a).length===Object.keys(b).length && Object.keys(a).every(k=>Object.hasOwn(b,k)&&sameData(a[k],b[k]));
const bindingKey = view => [view.snapshot.document.id,view.snapshot.document.sourceJSON,view.snapshot.document.floorBinding,view.snapshot.document.finishBinding];
const actionKey = view => [view.snapshot.expected,view.presentation.command,view.presentation.review?.reviewToken??null,view.presentation.retainedApplyRequest,view.presentation.outcome,view.presentation.actions];
export function readTileAuthoringUIView(input) {
  const value = finiteUIData(input);
  tileAuthoringExpected(value.snapshot);
  const p = value.presentation;
  if (!p || !p.actions || !Array.isArray(p.targetOwnerIds) || !Array.isArray(p.selectedCaptureMarkIds) || p.includeSelectedDecorationDefaults !== false) throw issue('UI_PRESENTATION_REQUIRED','Supply the current EditorDraftView; C1 decoration capture is unavailable.');
  for (const entry of Object.values(p.actions)) if (!entry || typeof entry.enabled !== 'boolean' || !Array.isArray(entry.diagnostics)) throw issue('UI_ACTION_FLAGS_REQUIRED','Editor supplies explicit enabled flags and diagnostics.');
  if (p.command !== null && (p.command?.kind !== 'nautilus-tile-authoring-command' || p.command.version !== 1)) throw issue('UI_CANONICAL_COMMAND_REQUIRED','No UI-intent or historical packet aliases.');
  if (p.outcome !== null && (!p.outcome || !['committed','rejected','unknown'].includes(p.outcome.outcome) || p.outcome.committed !== ({committed:true,rejected:false,unknown:null})[p.outcome.outcome] || typeof p.outcome.requestId !== 'string')) throw issue('UI_OUTCOME_REQUIRED','Keep committed true, rejected false and unknown null distinct.');
  if (p.outcome?.outcome === 'committed' && !['ready','pending','failed'].includes(p.outcome.presentation)) throw issue('UI_PRESENTATION_REQUIRED','Committed work survives failed presentation.');
  if (p.outcome?.outcome === 'unknown' && (p.outcome.committed !== null || p.outcome.writesFrozen !== true || p.outcome.recoveryRequestId !== p.outcome.requestId)) throw issue('UI_RECOVERY_ID_REQUIRED','Freeze writes and reconcile the same original request.');
  if (p.retainedApplyRequest !== null) {
    const request = p.retainedApplyRequest;
    if (!request || Object.keys(request).sort().join(',') !== 'expectedTargetDigest,requestId,reviewToken' ||
        typeof request.requestId !== 'string' || !/^[a-f0-9]{64}$/.test(request.reviewToken) || !/^[a-f0-9]{64}$/.test(request.expectedTargetDigest) || !p.review || request.requestId !== p.review.requestId || request.reviewToken !== p.review.reviewToken) throw issue('UI_RETAINED_REQUEST_REQUIRED','Use the request retained by this Editor connection.');
  }
  if (value.commandJSON !== undefined && value.commandJSON !== null && (typeof value.commandJSON !== 'string' || !p.command || !sameData(finiteUIData(JSON.parse(value.commandJSON)),p.command))) throw issue('UI_COMMAND_BYTES_REQUIRED','The retained command bytes must describe the actual staged command.');
  return value;
}

// Protocol/presentation coordinator only. No journal, private prepared handle or
// authority is owned here; every action is forwarded to borrowed Editor ports.
export function createTileAuthoringUIController(binding,{onView=()=>{},onPending=()=>{},onError=()=>{},readAdditionalData=null,isAdditionalConnectionCurrent=null,strictCallbackReads=false}={}) {
  const additional=readAdditionalData!==null||isAdditionalConnectionCurrent!==null||strictCallbackReads;
  if(additional&&(typeof readAdditionalData!=='function'||typeof isAdditionalConnectionCurrent!=='function'||strictCallbackReads!==true))throw issue('UI_ADDITIONAL_BINDING_REQUIRED','Bind both additional-source checks and strict callback reads together.');
  const service=Object.getOwnPropertyDescriptor(binding??{},'service')?.value;
  if (!binding || !serviceMethods.every(name=>ownFunction(service,name)) ||
      !['getSnapshot','subscribe','collectCommand','discardDraft'].every(name=>ownFunction(binding,name))) throw issue('UI_BINDING_REQUIRED','Supply one actual HostService and the Editor draft/view ports.');
  const collectCommand=Object.getOwnPropertyDescriptor(binding,'collectCommand').value;
  const originalMethods=serviceMethods.map(name=>Object.getOwnPropertyDescriptor(service,name).value);
  const borrowed=additional?['getSnapshot','subscribe','discardDraft'].map(name=>[name,Object.getOwnPropertyDescriptor(binding,name).value]):[];
  const additionalValues=new WeakMap();
  const capture=view=>{if(additional)additionalValues.set(view,finiteUIData(readAdditionalData(view)));return view;};
  const sameView=(a,b)=>sameData(actionKey(a),actionKey(b))&&sameData(bindingKey(a),bindingKey(b))&&(!additional||sameData(additionalValues.get(a),additionalValues.get(b)));
  const sameConnection=()=>{
    if(additional)try{if(isAdditionalConnectionCurrent()!==true)return false;}catch{return false;}
    if(Object.getOwnPropertyDescriptor(binding,'collectCommand')?.value!==collectCommand||Object.getOwnPropertyDescriptor(binding,'service')?.value!==service||!serviceMethods.every((name,i)=>Object.getOwnPropertyDescriptor(service,name)?.value===originalMethods[i]))return false;
    if(!additional)return true;
    return borrowed.every(([name,value])=>Object.getOwnPropertyDescriptor(binding,name)?.value===value);
  };
  let disposed=false,broken=false,lastView=null,identity=null,epoch=0,ticket=0,pending=false,rendering=false,queued=null,stop=null;
  const notify = reason => {try{onError(reason);}catch{}};
  const invalidate = reason => {lastView=null;try{onView(null);}catch{}notify(reason);};
  const read = () => {
    if(!sameConnection()){broken=true;ticket++;pending=false;throw issue('UI_REBIND_REQUIRED','The borrowed HostService connection changed.');}
    const raw=binding.getSnapshot(),view=capture(readTileAuthoringUIView(raw));
    if(additional&&!sameConnection()){broken=true;ticket++;pending=false;throw issue('UI_REBIND_REQUIRED','The borrowed connection changed during its snapshot callback.');}
    return {raw,view};
  };
  function receive(raw) {
    if(disposed||broken)return;if(!sameConnection()){broken=true;ticket++;pending=false;invalidate(issue('UI_REBIND_REQUIRED','The borrowed HostService connection changed.'));return;}queued={raw};if(rendering)return;rendering=true;let visits=0;
    try {
      while(queued&&!disposed&&!broken){
        if(++visits>8){queued=null;invalidate(issue('UI_OBSERVER_BOUND','Editor observer reentry exceeded the UI bound.'));break;}
        const {raw}=queued;queued=null;lastView=null;epoch++;
        try {
          const view=capture(readTileAuthoringUIView(raw)),next=bindingKey(view);
          if(additional&&!sameConnection()){broken=true;ticket++;pending=false;throw issue('UI_REBIND_REQUIRED','The borrowed connection changed during its source callback.');}
          if(identity!==null&&!sameData(next,identity)){broken=true;ticket++;pending=false;onPending(false);throw issue('UI_REBIND_REQUIRED','Workspace/floor/finish identity changed. Mount a new explicit Editor connection.');}
          onView(view);
          if(!queued&&!disposed&&!broken){identity=next;lastView=view;onPending(pending);}
        }catch(reason){invalidate(reason);}
      }
    }finally{rendering=false;}
  }
  function refresh(){if(disposed||broken)return null;try{receive(binding.getSnapshot());return lastView;}catch(reason){invalidate(reason);return null;}}
  async function perform(name,invoke) {
    const displayed=lastView,stamp=epoch;
    if(disposed||broken||rendering||!displayed||!enabled(displayed,name)||(pending&&name!=='cancel'))return {ok:false,code:'UI_ACTION_UNAVAILABLE'};
    let current;
    try{current=read();}catch(reason){invalidate(reason);return {ok:false,code:reason.code??'UI_VIEW_UNAVAILABLE'};}
    if(disposed||broken||epoch!==stamp||!sameView(current.view,displayed)||!sameData(bindingKey(current.view),identity)){
      receive(current.raw);return {ok:false,code:'UI_DISPLAY_CHANGED'};
    }
    if(!enabled(current.view,name)||(current.view.presentation.outcome?.outcome==='unknown'&&name!=='recover'))return {ok:false,code:'UI_ACTION_UNAVAILABLE'};
    const operation=++ticket;pending=true;
    try {
      onPending(true);
      if(!sameConnection()){broken=true;ticket++;pending=false;invalidate(issue('UI_REBIND_REQUIRED','The borrowed HostService connection changed.'));onPending(false);return {ok:false,code:'UI_REBIND_REQUIRED'};}
      if(disposed||broken||operation!==ticket||epoch!==stamp||!lastView)return {ok:false,code:'UI_DISPLAY_CHANGED'};
      if(additional){
        const checked=read();
        if(disposed||broken||operation!==ticket||epoch!==stamp||lastView!==displayed||!sameView(checked.view,current.view)){receive(checked.raw);return {ok:false,code:'UI_DISPLAY_CHANGED'};}
      }
      const canCollect=()=>{
        if(disposed||broken||rendering||operation!==ticket||epoch!==stamp||lastView!==displayed)return false;
        if(!sameConnection()){broken=true;ticket++;pending=false;invalidate(issue('UI_REBIND_REQUIRED','The borrowed Editor connection changed during construction.'));onPending(false);return false;}
        if(additional){
          const checked=read();
          if(disposed||broken||rendering||operation!==ticket||epoch!==stamp||lastView!==displayed||!sameConnection()||!sameView(checked.view,displayed))return false;
        }
        return enabled(lastView,name)&&lastView.presentation.outcome?.outcome!=='unknown';
      };
      if(additional){
        if(!sameConnection()){broken=true;ticket++;pending=false;invalidate(issue('UI_REBIND_REQUIRED','The borrowed connection changed before invocation.'));return {ok:false,code:'UI_REBIND_REQUIRED'};}
        if(disposed||broken||operation!==ticket||epoch!==stamp||lastView!==displayed)return {ok:false,code:'UI_DISPLAY_CHANGED'};
      }
      const result=await invoke(current.raw,current.view,canCollect);
      if(!disposed&&!broken&&operation===ticket&&result?.code!=='UI_STAGE_INVALIDATED')refresh();
      return result;
    }catch(reason){if(!disposed&&!broken&&operation===ticket)notify(reason);return {ok:false,code:reason.code??'UI_ACTION_FAILED',message:reason.message};}
    finally{if(!disposed&&!broken&&operation===ticket){pending=false;onPending(false);}}
  }
  const api = {
    // UI intent guard only. Reuse this coordinator's displayed identity, epoch,
    // strict callback read and complete additional-data comparison. Never execute
    // a command or turn a displayed/parsed copy into an actual issued read.
    captureResourceDisplay(displayed){
      const stamp=epoch;
      const stable=()=>!disposed&&!broken&&!rendering&&lastView===displayed&&epoch===stamp;
      const changed=raw=>{if(!disposed&&!broken){if(raw===undefined)refresh();else receive(raw);}throw issue('UI_DISPLAY_CHANGED','The displayed source, selection or action state changed. Inspect the refreshed editor before loading artwork.');};
      if(!displayed||!stable())return changed();
      const guard=Object.freeze({read(){
        if(!stable())return changed();
        let checked;try{checked=read();}catch(reason){invalidate(reason);throw reason;}
        if(!stable()||!sameView(checked.view,displayed)||!sameData(bindingKey(checked.view),identity)||!sameData(checked.view.snapshot,displayed.snapshot)||!sameData(checked.view.presentation,displayed.presentation))return changed(checked.raw);
        // read() has completed all additional-source/currentness callbacks. Take
        // the final actual read now, compare its data, and call no user callback
        // between this comparison and returning it to the resource dispatcher.
        const raw=binding.getSnapshot(),view=readTileAuthoringUIView(raw);
        if(Object.getOwnPropertyDescriptor(binding,'collectCommand')?.value!==collectCommand||Object.getOwnPropertyDescriptor(binding,'service')?.value!==service||!serviceMethods.every((name,i)=>Object.getOwnPropertyDescriptor(service,name)?.value===originalMethods[i])||!borrowed.every(([name,value])=>Object.getOwnPropertyDescriptor(binding,name)?.value===value)){
          broken=true;ticket++;pending=false;const reason=issue('UI_REBIND_REQUIRED','The borrowed connection changed during the final resource read.');invalidate(reason);throw reason;
        }
        if(!stable()||!sameData(actionKey(view),actionKey(displayed))||!sameData(bindingKey(view),bindingKey(displayed))||!sameData(view.snapshot,displayed.snapshot)||!sameData(view.presentation,displayed.presentation)||!sameData(view.overlay,displayed.overlay))return changed(raw);
        return raw;
      }});
      guard.read();return guard;
    },
    get pending(){return pending;},
    get view(){return lastView;},
    refresh,
    stage(kind,build){return perform(kind,(raw,view,canCollect)=>{
      const command=build(view.snapshot,view,raw);
      if(!canCollect())return {ok:false,code:'UI_STAGE_INVALIDATED',message:'Editor state changed during draft creation. Check the current editor state.'};
      return collectCommand.call(binding,command);
    });},
    review(){return perform('review',raw=>{if(!raw.presentation.command)throw issue('UI_COMMAND_REQUIRED','Stage a finite canonical command first.');return service.prepare(raw.presentation.command);});},
    apply(){return perform('apply',raw=>{if(!raw.presentation.retainedApplyRequest)throw issue('UI_RETAINED_REQUEST_REQUIRED','Review through this connection first.');return service.apply(raw.presentation.retainedApplyRequest);});},
    cancel(){return perform('cancel',raw=>raw.presentation.review?service.cancel(raw.presentation.review.reviewToken):binding.discardDraft());},
    recover(){return perform('recover',raw=>{const result=raw.presentation.outcome;if(result?.outcome!=='unknown')throw issue('UI_RECOVERY_ID_REQUIRED','Use the original unknown request.');return service.recover(result.recoveryRequestId);});},
    history(name,...args){return perform(name,()=>{if(!['undo','redo','save','open','retryPresentation'].includes(name)||!ownFunction(binding.history,name))throw issue('UI_HISTORY_UNBOUND','Use existing Editor history/save/presentation ports.');return binding.history[name](...args);});},
    dispose(){if(disposed)return;disposed=true;ticket++;pending=false;lastView=null;queued=null;const unsubscribe=stop;stop=null;unsubscribe?.();}
  };
  try{const unsubscribe=binding.subscribe(receive);if(typeof unsubscribe!=='function')throw issue('UI_SUBSCRIPTION_REQUIRED','Editor subscription must return its disposer.');if(disposed)unsubscribe();else stop=unsubscribe;refresh();}catch(reason){api.dispose();throw reason;}
  return Object.freeze(api);
}

// One explicit attachment to the existing coordinator, service and history.
// Detaching this controller removes only its own observer and pending callbacks.
export function attachTileAuthoringOverlayCoordinator(binding,options){
  if(options?.strictCallbackReads!==true||typeof options.readAdditionalData!=='function'||typeof options.isAdditionalConnectionCurrent!=='function')throw issue('UI_ADDITIONAL_BINDING_REQUIRED','The overlay attachment requires its complete source/currentness comparison.');
  return createTileAuthoringUIController(binding,options);
}

  function showDiagnostics(){const p=current?.presentation,all=[...(Array.isArray(p?.review?.diagnostics)?p.review.diagnostics:[]),...(p?.outcome?.diagnostics??[]),...attemptedDiagnostics],messages=new Map();for(const d of all){const key=diagnosticText(d,false);if(!messages.has(key))messages.set(key,diagnosticText(d));}fault.textContent=[...messages.values()].join('\n');fault.hidden=messages.size===0;}
  function showError(reason){attemptedDiagnostics=[{code:reason?.code,message:reason?.message??String(reason)}];showDiagnostics();try{onError(reason);}catch{}}
  function showResult(result){if(result?.diagnostics?.length){attemptedDiagnostics=result.diagnostics;showDiagnostics();}else if(result?.ok===false&&result.code&&!['UI_ACTION_UNAVAILABLE','UI_DISPLAY_CHANGED'].includes(result.code))showError(issue(result.code,result.message??'The host refused this action.'));}
  function identities(kind){return finiteUIData(binding.allocateIds(kind));}
  function selectedDefinition(){const result=definitionMap.get(definitionSelect.value);if(!result)throw issue('UI_DEFINITION_REQUIRED','Select an exact immutable definition revision.');return result;}
  function definitionByPin(pin){const result=definitionMap.get(pinKey(pin));if(!result)throw issue('UI_DEFINITION_REQUIRED','The exact retained definition pin is unavailable.');return result;}
  function selectedPreset(){const result=presetMap.get(presetSelect.value);if(!result)throw issue('UI_PRESET_REQUIRED','Select an exact immutable preset revision.');return result;}
  function actualFloor(snapshot){const id=binding.borrowedFloorController?.state?.selectedId,floor=snapshot.document.floorBinding.floors.find(f=>f.id===id);if(!floor)throw issue('UI_FLOOR_UNBOUND','Borrow the current declared floor controller.');return floor;}
  function options(node,rows,placeholder){const signature=stringify(rows);if(selectSignatures.get(node)===signature)return;const selected=node.value;node.replaceChildren(new doc.defaultView.Option(placeholder,''),...rows.map(r=>{const option=new doc.defaultView.Option(r.label,r.key);option.title=r.title??r.key;return option;}));node.value=rows.some(r=>r.key===selected)?selected:'';selectSignatures.set(node,signature);}
  const revisionName=record=>record.body.label+' r'+record.revision;
  function definitionName(pin){const record=definitionMap.get(pinKey(pin));return record?revisionName(record):'retained revision r'+pin.revision;}
  function ownerLabel(ownerId){const document=current?.snapshot.document,index=document?.authoring.ownerPins.findIndex(p=>p.ownerId===ownerId),owner=document?.tileBundle.layout.instances.find(o=>o.id===ownerId);return owner?.label||((index??-1)>=0?'Copy '+(index+1):'Selected copy');}
  function selectionText(){
    definitionPin.textContent=definitionMap.has(definitionSelect.value)?'Definition '+definitionSelect.value:'No definition selected.';presetPin.textContent=presetMap.has(presetSelect.value)?'Preset '+presetSelect.value:'No preset selected.';
    definitionSelect.title=definitionMap.has(definitionSelect.value)?definitionSelect.value:'';presetSelect.title=presetMap.has(presetSelect.value)?presetSelect.value:'';ownerSelect.title=ownerSelect.value;
    const pin=current?.snapshot.document.authoring.ownerPins.find(p=>p.ownerId===ownerSelect.value),owner=current?.snapshot.document.tileBundle.layout.instances.find(o=>o.id===ownerSelect.value),target=definitionMap.get(definitionSelect.value);
    ownerPinText.textContent=pin?'Owner '+pin.ownerId+' · From '+stringify(pin.definition):'No upgrade target selected.';
    ownerInfo.textContent=pin?ownerLabel(pin.ownerId)+' · From '+definitionName(pin.definition)+(target?' → '+revisionName(target):'. Choose a target definition revision.')+(owner?.locked?' · locked':''):'Select a copy to see its upgrade.';
    updateGates();
  }
  function controlReason(action,allowed){
    if(allowed)return '';
    if(current?.presentation.outcome?.outcome==='unknown')return 'Check the original edit result before editing.';
    if(controller?.pending)return 'Wait for the current action.';
    if(['recipe.clone','recipe.revise','preset.save','placement.upgrade'].includes(action)&&!definitionMap.has(definitionSelect.value))return 'Select a definition revision.';
    if(action==='preset.place'&&!presetMap.has(presetSelect.value))return 'Select a preset revision.';
    if(action==='placement.upgrade'&&!ownerSelect.value)return 'Select a copy to upgrade.';
    if(action==='placement.upgrade'&&current?.snapshot.document.tileBundle.layout.instances.find(o=>o.id===ownerSelect.value)?.locked)return 'This copy is locked.';
    return current?.presentation.actions[action]?.diagnostics[0]?.message??'';
  }
  function scopeText(document,p){
    const operation=p.command?.operations[0],body=operation?.body;
    if(operation?.kind==='placement.upgrade')return body.ownerIds.map(ownerLabel).join(', ')+' · From '+definitionName(body.from)+' → '+definitionName(body.to);
    if(operation?.kind==='preset.place'){const preset=presetMap.get(pinKey(body.preset));return 'Place '+body.placements.length+' independent copies'+(preset?' · '+revisionName(preset):'');}
    if(operation?.kind==='preset.save')return 'Reusable preset · '+revisionName(body.preset)+'. Existing copies stay unchanged.';
    if(operation?.kind?.startsWith('recipe.'))return revisionName(body.definition)+'. Placed copies stay unchanged.';
    return document.authoring.ownerPins.length+' placed '+(document.authoring.ownerPins.length===1?'copy':'copies')+' · content revision '+document.contentRevision;
  }
  function updateGates(){for(const {node,action}of gates){let allowed=!!current&&!!controller?.view&&enabled(current,action)&&(!controller?.pending||action==='cancel');if(['recipe.clone','recipe.revise','preset.save','placement.upgrade'].includes(action)&&!definitionMap.has(definitionSelect.value))allowed=false;if(action==='preset.place'&&!presetMap.has(presetSelect.value))allowed=false;if(action==='placement.upgrade'&&(!ownerSelect.value||current?.snapshot.document.tileBundle.layout.instances.find(o=>o.id===ownerSelect.value)?.locked))allowed=false;if(current?.presentation.outcome?.outcome==='unknown'&&action!=='recover')allowed=false;node.disabled=!allowed;node.title=controlReason(action,allowed);if(action==='recover')node.hidden=current?.presentation.outcome?.outcome!=='unknown';if(action==='retryPresentation')node.hidden=current?.presentation.outcome?.outcome!=='committed'||current.presentation.outcome.presentation==='ready';}
    for(const input of root.querySelectorAll('input,select,textarea'))input.disabled=!current||!!controller?.pending||current.presentation.outcome?.outcome==='unknown';
  }
  function render(view){
    const previous=current?.presentation,next=view?.presentation;
    if(previous&&next&&(previous.command?.requestId!==next.command?.requestId||previous.review?.reviewToken!==next.review?.reviewToken||previous.outcome?.requestId!==next.outcome?.requestId||previous.outcome?.outcome!==next.outcome?.outcome))attemptedDiagnostics=[];
    current=view;if(!view){status.textContent='Editor connection unavailable';updateGates();return;}
    const {document}=view.snapshot,p=view.presentation;definitionMap=new Map(document.authoring.library.definitions.map(r=>[pinKey(r),r]));presetMap=new Map(document.authoring.library.presets.map(r=>[pinKey(r),r]));
    options(definitionSelect,[...definitionMap].map(([key,r])=>({key,label:r.body.label+' · r'+r.revision})),'Select a definition revision');
    options(presetSelect,[...presetMap].map(([key,r])=>({key,label:r.body.label+' · r'+r.revision})),'Select a preset revision');
    options(ownerSelect,document.authoring.ownerPins.map(p=>{const owner=document.tileBundle.layout.instances.find(o=>o.id===p.ownerId);return{key:p.ownerId,label:ownerLabel(p.ownerId)+(owner?.locked?' · locked':'')};}),'Select one upgrade target');
    status.textContent=p.outcome?.outcome==='unknown'?'Edit result unknown — recover the original request':p.review?'Review ready — inspect before Apply':p.command?'Draft — review required':p.outcome?.outcome==='committed'?'Saved; presentation '+p.outcome.presentation:p.outcome?.outcome==='rejected'?'Edit refused; accepted work retained':'Ready';
    scope.textContent=scopeText(document,p);technicalScope.textContent='Scope: '+p.scope+' · Targets: '+(p.targetOwnerIds.join(', ')||'library only')+' · Content revision: '+document.contentRevision;
    preparation.hidden=!p.review;preparation.textContent=p.review?.tilePreparation==='source-unchanged'?'Placed geometry stays unchanged.':p.review?.tilePreparation==='public-v1-prepared'?'Native rectangle preview ready.':'';
    diagnostics.textContent=JSON.stringify({actions:p.actions,reviewDiagnostics:p.review?.diagnostics??[],outcomeDiagnostics:p.outcome?.diagnostics??[]},null,2);showDiagnostics();
    readiness.replaceChildren(...Object.entries(view.readiness??{authoring:{state:'not-reported'},visual:{state:'not-reported'},game:{state:'unbound'}}).flatMap(([name,value])=>[make('dt',name),make('dd',typeof value==='string'?value:stringify(value))]));
    detailText.textContent=view.commandJSON??'No retained command bytes are reported by Editor.';
    let floor;try{floor=actualFloor(view.snapshot);}catch{}placementFloor.textContent=floor?'Copies use drawing floor '+floor.label+' · '+floor.elevationM+' m.':'Placement floor binding is unavailable.';
    selectionText();
  }
  for(const node of [definitionSelect,presetSelect,ownerSelect])listen(node,'change',selectionText);
  listen(root,'keydown',event=>{if(event.repeat&&['Enter',' '].includes(event.key)&&event.target.closest?.('button,form'))event.preventDefault();});
  try{controller=createTileAuthoringUIController(binding,{onView:render,onPending:updateGates,onError:showError});updateGates();}catch(reason){events.abort();root.remove();throw reason;}
  return Object.freeze({element:root,refresh:()=>controller.refresh(),dispose(){const errors=[];for(const cleanup of [()=>controller.dispose(),()=>events.abort(),()=>root.remove()])try{cleanup();}catch(reason){errors.push(reason);}return errors;}});
}
